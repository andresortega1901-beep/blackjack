<script>

let shoe = []; 

let globalLog = []; 


let numSeats = 7;

let selectedTarget = 'seat-1-0'; 

let revealingHiddenDealerCard = false;

let initialDealOrder = [];

let initialDealIndex = 0;

let tableState = {

  dealer: [],

  seats: {}

};


// Variables para el reconocimiento de voz

let recognition = null;

let isListening = false;


function initTableState() {

  for (let i = 1; i <= 7; i++) {

    tableState.seats[i] = {

      hands: [[]],

      activeSub: 0

    };

  }

}

initTableState();


function hilo(rank) {

  if (['2','3','4','5','6'].includes(rank)) return 1;

  if (['10','A'].includes(rank)) return -1;

  return 0;

}

function el(id){ return document.getElementById(id); }


function initApp() {

  updateSeatsCount();

  renderTable();

  renderCounters();

  initVoiceRecognition();

}


// Configuración de la API de Voz

function initVoiceRecognition() {

  const SpeechRecognition = window.SpeechRecognition || window.webkitSpeechRecognition;

  if (!SpeechRecognition) {

    el('voiceStatusText').textContent = 'Voz no soportada en este navegador';

    el('voiceMicBtn').style.opacity = '0.5';

    el('voiceMicBtn').disabled = true;

    return;

  }


  recognition = new SpeechRecognition();

  recognition.lang = 'es-ES';

  recognition.continuous = true;

  recognition.interimResults = false;


  recognition.onresult = (event) => {

    const transcript = event.results[event.results.length - 1][0].transcript.trim().toLowerCase();

    parseVoiceCommand(transcript);

  };


  recognition.onend = () => {

    if (isListening) {

      try { recognition.start(); } catch(e) {}

    }

  };

}


function toggleVoiceRecognition() {

  if (!recognition) return;

  const btn = el('voiceMicBtn');

  const text = el('voiceStatusText');


  if (!isListening) {

    try {

      recognition.start();

      isListening = true;

      btn.classList.add('listening');

      text.textContent = 'Escuchando cartas... (ej. "As", "Diez", "Rey")';

    } catch(e) {

      console.error(e);

    }

  } else {

    isListening = false;

    recognition.stop();

    btn.classList.remove('listening');

    text.textContent = 'Activar Comando por Voz (Micrófono)';

  }

}


function parseVoiceCommand(spokenText) {

  let cleaned = spokenText.replace(/[.,\/#!$%\^&\*;:{}=\-_`~()?]/g,"").trim();

  const voiceMap = {

    'as': 'A', 'haz': 'A', '1': 'A',

    'dos': '2', '2': '2',

    'tres': '3', '3': '3',

    'cuatro': '4', '4': '4',

    'cinco': '5', '5': '5',

    'seis': '6', '6': '6',

    'siete': '7', '7': '7',

    'ocho': '8', '8': '8',

    'nueve': '9', '9': '9',

    'diez': '10', '10': '10', 'jota': 'J', 'j': '10', 'reina': 'Q', 'q': '10', 'rey': 'K', 'king': 'K', 'k': '10'

  };


  const words = cleaned.split(/\s+/);

  for (let w of words) {

    if (voiceMap[w]) {

      let val = voiceMap[w];

      if(['J','Q','K'].includes(val)) val = '10'; // Traduce figuras automáticamente a 10

      handlePickedCard(val);

      break;

    }

  }

}


function updateSeatsCount() {

  numSeats = Number(el('numSeats').value);

  buildInitialDealOrder();

  renderTable();

  updateTargetIndicator();

}


function buildInitialDealOrder() {

  const oneRound = [];

  for (let i = 1; i <= numSeats; i++) oneRound.push(`seat-${i}-0`);

  oneRound.push('dealer');

  initialDealOrder = [...oneRound, ...oneRound];

  initialDealIndex = 0;

  selectedTarget = 'seat-1-0';

}


function updateTargetIndicator() {

  const indicator = el('targetIndicator');

  if (initialDealIndex < initialDealOrder.length) {

    const target = initialDealOrder[initialDealIndex];

    selectedTarget = target;

    if (target === 'dealer') {

      indicator.textContent = `Reparto automático ${initialDealIndex + 1}/${initialDealOrder.length}: DEALER`;

    } else {

      const parts = target.split('-');

      indicator.textContent = `Reparto automático ${initialDealIndex + 1}/${initialDealOrder.length}: PUESTO ${parts[1]}`;

    }

  } else {

    indicator.textContent = 'Selecciona un puesto o submano para continuar.';

  }

}


function selectTarget(target) {

  selectedTarget = target;

  if (target === 'dealer') {

    el('targetIndicator').textContent = 'Seleccionado: DEALER';

  } else {

    const parts = target.split('-');

    const seatNum = parts[1];

    const subIdx = parts[2] || 0;

    tableState.seats[seatNum].activeSub = Number(subIdx);

    el('targetIndicator').textContent = `Seleccionado: PUESTO ${seatNum} (Mano ${Number(subIdx)+1})`;

  }

  renderTable();

}


function assignHiddenDealerCard() {

  const decks = Number(el('decks').value);

  if (shoe.length >= decks * 52) { alert('El zapato está completo.'); return; }


  const isAutomaticDeal = initialDealIndex < initialDealOrder.length;

  const autoIndexBefore = isAutomaticDeal && initialDealOrder[initialDealIndex] === 'dealer' ? initialDealIndex : null;


  shoe.push('??');

  globalLog.push({ target: 'dealer', card: '??', autoIndexBefore });

  tableState.dealer.push('??');


  if (autoIndexBefore !== null) {

    initialDealIndex++;

    updateTargetIndicator();

  } else {

    selectedTarget = 'dealer';

    el('targetIndicator').textContent = 'Carta oculta asignada al DEALER';

  }


  renderTable();

  renderCounters();

}


function revealHiddenDealerCard() {

  const hiddenIndex = tableState.dealer.lastIndexOf('??');

  if (hiddenIndex === -1) { alert('No hay ninguna carta oculta del Dealer para revelar.'); return; }

  revealingHiddenDealerCard = true;

  selectedTarget = 'dealer';

  el('targetIndicator').textContent = 'Revelando carta oculta: selecciona su valor';

  renderTable();

}


function handlePickedCard(rank) {

  if (revealingHiddenDealerCard) {

    const dealerIndex = tableState.dealer.lastIndexOf('??');

    const logIndex = globalLog.map(x => x).lastIndexOf(

      globalLog.slice().reverse().find(x => x.target === 'dealer' && x.card === '??')

    );

    if (dealerIndex === -1 || logIndex === -1) {

      revealingHiddenDealerCard = false;

      alert('No se encontró la carta oculta que se desea revelar.');

      return;

    }

    tableState.dealer[dealerIndex] = rank;

    globalLog[logIndex].card = rank;

    shoe[logIndex] = rank;

    revealingHiddenDealerCard = false;

    el('targetIndicator').textContent = 'Carta del Dealer revelada';

    renderTable();

    renderCounters();

    return;

  }

  assignCardToTarget(rank);

}


function assignCardToTarget(rank) {

  const decks = Number(el('decks').value);

  if(shoe.length >= decks * 52) { alert('El zapato está completo.'); return; }


  const isAutomaticDeal = initialDealIndex < initialDealOrder.length;

  const target = isAutomaticDeal ? initialDealOrder[initialDealIndex] : selectedTarget;

  const indexBefore = isAutomaticDeal ? initialDealIndex : null;


  shoe.push(rank);

  globalLog.push({ target, card: rank, autoIndexBefore: indexBefore });


  if(target === 'dealer') {

    tableState.dealer.push(rank);

  } else {

    const parts = target.split('-');

    const seatNum = parts[1];

    const subIdx = Number(parts[2] || 0);

    tableState.seats[seatNum].hands[subIdx].push(rank);


    if (tableState.seats[seatNum].hands.length === 2 && subIdx === 0 && tableState.seats[seatNum].hands[0].length === 2) {

      selectedTarget = `seat-${seatNum}-1`;

    }

  }


  if (isAutomaticDeal) {

    initialDealIndex++;

    updateTargetIndicator();

  }


  renderTable();

  renderCounters();

}


function splitHand(seatNum) {

  const seat = tableState.seats[seatNum];

  if (seat.hands.length >= 2) return; 

  const currentHand = seat.hands[0];

  if (currentHand.length !== 2 || currentHand[0] !== currentHand[1]) return;


  const card1 = currentHand[0];

  const card2 = currentHand[1];


  seat.hands = [

    [card1],

    [card2]

  ];


  selectedTarget = `seat-${seatNum}-0`;

  renderTable();

}


function undoLastGlobalCard() {

  if(globalLog.length === 0) return;

  const last = globalLog.pop();

  shoe.pop();


  if(last.target === 'dealer') {

    if (tableState.dealer.length) tableState.dealer.pop();

  } else {

    const parts = last.target.split('-');

    const seatNum = parts[1];

    const subIdx = Number(parts[2] || 0);

    if (tableState.seats[seatNum].hands[subIdx].length) {

      tableState.seats[seatNum].hands[subIdx].pop();

    }

  }


  if (last.autoIndexBefore !== null && last.autoIndexBefore !== undefined) {

    initialDealIndex = last.autoIndexBefore;

    updateTargetIndicator();

  }


  renderTable();

  renderCounters();

}


function clearRound() {

function renderTable() {

  el('dealerCards').innerHTML = tableState.dealer.length ? tableState.dealer.map(c => c === '??' ? '<div class="card-chip hidden-card">🂠 Oculta</div>' : `<div class="card-chip">${c}</div>`).join('') : '<span class="muted" style="font-size:12px;">Sin cartas</span>';


  const rules = { das: el('das').value==='yes', h17: el('soft17').value==='H17', surrender: el('surrender').checked };

  const dealerCard = tableState.dealer.find(c => c !== '??') || null;


  let html = '';

  for(let i=1; i<=numSeats; i++) {

    const seat = tableState.seats[i];

    const cardsHtmlList = seat.hands.map((hand, subIdx) => {

      const targetKey = `seat-${i}-${subIdx}`;

      const isSelected = selectedTarget === targetKey;

      const handCardsHtml = hand.length ? hand.map(c => `<div class="card-chip">${c}</div>`).join('') : '<span class="muted" style="font-size:11px;">Vacío</span>';

      

      let seatActionText = '—';

      let seatActionColor = 'var(--muted)';

      if (hand.length >= 2 && dealerCard) {

        const res = decide(hand, dealerCard, rules);

        seatActionText = actionName(res.a);

        seatActionColor = res.a==='HIT'?'var(--amber)':res.a==='STAND'?'var(--green)':res.a==='DOUBLE'?'var(--green)':res.a==='SPLIT'?'var(--blue)':'var(--red)';

      } else {

        seatActionText = hand.length < 2 ? 'Faltan cartas' : 'Falta crupier';

      }


      const canSplit = hand.length === 2 && hand[0] === hand[1] && seat.hands.length < 2;


      return `

        <div class="${subIdx > 0 ? 'split-subhand' : ''}" onclick="selectTarget('${targetKey}')">

          <div style="font-size:10px; color:${isSelected?'var(--green)':'var(--muted)'}; font-weight:bold; margin-bottom:2px;">

            ${seat.hands.length > 1 ? 'Mano ' + (subIdx+1) : 'Mano principal'} ${isSelected ? '●' : ''}

          </div>

          <div class="cards-container">${handCardsHtml}</div>

          <div style="font-size:10px; color:var(--muted); display:flex; justify-content:space-between;">

            <span>Total: ${hand.length ? handInfo(hand).total : 0}</span>

          </div>

          <div class="seat-optimal-action" style="color:${seatActionColor};">${seatActionText}</div>

          ${canSplit ? `<button type="button" class="split-btn" onclick="event.stopPropagation(); splitHand(${i})">Dividir (Split)</button>` : ''}

        </div>

      `;

    }).join('');


    html += `

      <div class="seat-box">

        <div>

          <div class="seat-header">

            <span>Puesto ${i}</span>

          </div>

          ${cardsHtmlList}

        </div>

      </div>

    `;

  }

  el('seatsGrid').innerHTML = html;

}


function renderCounters() {

  const rc = shoe.reduce((s,c)=>s+hilo(c),0);

  const decks = Number(el('decks').value);

  const totalCardsInShoe = decks * 52;

  const seen = shoe.length; 

  const remainingCards = Math.max(0, totalCardsInShoe - seen);

  const remainingDecks = remainingCards / 52;

  const tc = remainingDecks > 0 ? rc / remainingDecks : 0;

  

  el('rc').textContent = rc > 0 ? '+' + rc : rc;

  el('tc').textContent = remainingDecks > 0 ? tc.toFixed(2) : '—';

  

  const estEdge = -0.40 + (tc * 0.5);

  el('betEdge').textContent = `${estEdge > 0 ? '+' : ''}${estEdge.toFixed(2)}%`;

  el('betEdge').className = estEdge > 0 ? 'green' : 'red';

  el('betSignal').textContent = estEdge > 0.5 ? 'Subir apuesta' : 'Mantener mínima';

  el('betSignal').className = estEdge > 0.5 ? 'green' : 'amber';


  el('remaining').textContent = `Barajas restantes: ${remainingDecks.toFixed(2)} · Cartas usadas: ${seen}/${totalCardsInShoe}`;

  el('history').textContent = shoe.length ? shoe.map((c,i)=>`${i+1}:${c === '??' ? 'Oculta' : c}`).join(' · ') : 'Sin cartas registradas.';

  renderShoeAudit();

  renderTable(); 

}

  const decks = Number(el('decks').value);

  if (!audit) return;

  audit.innerHTML = `<strong class="audit-ok">Inventario consistente.</strong> Cartas en juego: ${shoe.length}.`;

}


function cardValue(rank){ if(rank==='A') return 11; return Number(rank); }

function handInfo(hand){

  let total = hand.reduce((s,c)=>s+cardValue(c),0), aces = hand.filter(c=>c==='A').length;

  while(total > 21 && aces > 0){ total -= 10; aces--; }

  const soft = hand.includes('A') && total <= 21 && hand.reduce((s,c)=>s+cardValue(c),0) === total;

  return {total, soft};

}

function actionName(a){ return ({HIT:'PEDIR', STAND:'PLANTARSE', DOUBLE:'DOBLAR', SPLIT:'DIVIDIR', SURRENDER:'RENDIRSE'})[a]||a; }


function decide(hand, dealerRank, rules){

  if(!dealerRank) return {a:'ESPERAR', why:'Asigna la carta visible del crupier.'};

  const d = dealerRank==='A' ? 11 : cardValue(dealerRank);

  const info = handInfo(hand), t = info.total, two = hand.length===2;

  const pair = two && hand[0]===hand[1];

  const das = rules.das, h17 = rules.h17, surrender = rules.surrender;

  

  const decks = Number(el('decks').value);

  const remainingDecks = Math.max(0, (decks*52 - shoe.length) / 52);

  const rc = shoe.reduce((s,c)=>s+hilo(c),0);

  const tc = remainingDecks > 0 ? rc / remainingDecks : 0;


  if(t > 21) return {a:'BUST', why:'La mano supera 21.'};


  if (two && !info.soft && !pair) {

    if (t === 16 && d === 10 && tc >= 0) return {a:'STAND', why:`Desviación TC (${tc.toFixed(1)} ≥ 0): Plantarse con 16 vs 10.`};

    if (t === 15 && d === 10 && tc >= 4) return {a:'STAND', why:`Desviación TC (${tc.toFixed(1)} ≥ 4): Plantarse con 15 vs 10.`};

    if (t === 12 && d === 3 && tc >= 2) return {a:'STAND', why:`Desviación TC (${tc.toFixed(1)} ≥ 2): Plantarse con 12 vs 3.`};

    if (t === 12 && d === 2 && tc >= 3) return {a:'STAND', why:`Desviación TC (${tc.toFixed(1)} ≥ 3): Plantarse con 12 vs 2.`};

    if (t === 10 && d === 11 && tc >= 4) return {a:'DOUBLE', why:`Desviación TC (${tc.toFixed(1)} ≥ 4): Doblar 10 vs As.`};

    if (t === 10 && d === 10 && tc >= 4) return {a:'DOUBLE', why:`Desviación TC (${tc.toFixed(1)} ≥ 4): Doblar 10 vs 10.`};

  }


  if(two && surrender && !pair && !info.soft && ((t===16 && [9,10,11].includes(d)) || (t===15 && d===10))) {

    return {a:'SURRENDER', why:'Rendición tardía aplicada.'};

  }

  if(pair){

    const p=hand[0], v=cardValue(p);

    if(p==='A' || p==='8') return {a:'SPLIT', why:'Dividir ases y ochos siempre.'};

    if(v===10) return {a:'STAND', why:'No dividir pares de 10.'};

    if(p==='9') return {a:([2,3,4,5,6,8,9].includes(d)?'SPLIT':'STAND'), why:'Estrategia para par de 9s.'};

    if(p==='7') return {a:d>=2&&d<=7?'SPLIT':'HIT', why:'Dividir 7s vs 2–7.'};

    if(p==='6') return {a:(d>=(das?2:3)&&d<=6)?'SPLIT':'HIT', why:'Dividir 6s.'};

    if(p==='4') return {a:(das&&[5,6].includes(d))?'SPLIT':'HIT', why:'Dividir 4s con DAS vs 5-6.'};

    if(p==='2'||p==='3') return {a:(d>=(das?2:4)&&d<=7)?'SPLIT':'HIT', why:'Dividir 2s y 3s.'};

  }

  if(info.soft){

    if(t>=19) return {a:'STAND', why:'17 suave o más: plantarse.'};

    if(t===18){

      if(two && d>=3&&d<=6) return {a:'DOUBLE', why:'18 suave: doblar vs 3–6.'};

      if(two && h17 && d===2) return {a:'DOUBLE', why:'18 suave dobla vs 2 (con H17).'};

      if(d===2) return {a:'STAND', why:'18 suave: plantarse vs 2.'};

      if(d===7||d===8) return {a:'STAND', why:'18 suave: plantarse vs 7–8.'};

      return {a:'HIT', why:'18 suave: pedir vs 9, 10 o As.'};

    }

    if(t===17 && two && d>=3 && d<=6) return {a:'DOUBLE', why:'17 suave: doblar vs 3–6.'};

    if((t===15||t===16) && two && d>=4 && d<=6) return {a:'DOUBLE', why:'Mano suave: doblar vs 4–6.'};

    if((t===13||t===14) && two && d>=5 && d<=6) return {a:'DOUBLE', why:'Mano suave: doblar vs 5–6.'};

    return {a:'HIT', why:'Mano suave baja: pedir carta.'};

  }

  if(t<=8) return {a:'HIT', why:'Total duro de 8 o menos: pedir.'};

  if(t===9) return {a:two&&d>=3&&d<=6?'DOUBLE':'HIT', why:'9 duro: doblar vs 3–6.'};

  if(t===10) return {a:two&&d>=2&&d<=9?'DOUBLE':'HIT', why:'10 duro: doblar vs 2–9.'};

  if(t===11) return {a:two&&(d<=10||h17&&d===11)?'DOUBLE':'HIT', why:'11 duro: doblar.'};

  if(t===12) return {a:d>=4&&d<=6?'STAND':'HIT', why:'12 duro: plantarse vs 4–6.'};

  if(t>=13&&t<=16) return {a:d>=2&&d<=6?'STAND':'HIT', why:'13–16 duro: plantarse vs 2–6.'};

  return {a:'STAND', why:'17 duro o más: plantarse.'};

}
